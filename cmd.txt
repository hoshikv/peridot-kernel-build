export PATH="$HOME/tc/bin:$PATH"
D="$GITHUB_WORKSPACE/display-drivers/vendor_opensource_display-drivers-peridot-u-oss"
L="$D/msm"
echo "=== external UNDEFINED (non-kernel) symbols referenced by built .o ==="
for f in $(find "$L" -name "*.o" | grep -v "\.mod\.o" | grep -v modules.o); do
  ~/tc/bin/llvm-nm -u "$f" 2>/dev/null | awk '{print $2}'
done | sort -u > /tmp/opencode/undef.txt
# filter out kernel-builtin-ish symbols: those defined by vmlinux Module.symvers are OK; external module ones come from sources not in import
echo "total undefined refs: $(wc -l < /tmp/opencode/undef.txt)"
echo "=== candidates that are external modules (hdcp/ext_disp/fence/spec/sec/qseecom mmrm) ==="
grep -iE "hdcp|ext_disp|hw_fence|spec_sync|mmrm|qseecom|sec_|sar|charger_detect|dsi_cmd" /tmp/opencode/undef.txt | head -60
