export PATH="$HOME/tc/bin:$PATH"
export ARCH=arm64 SUBARCH=arm64 CC=clang LLVM=1 LLVM_IAS=1
cd "$GITHUB_WORKSPACE/kernel"
O="$GITHUB_WORKSPACE/out"
D="$GITHUB_WORKSPACE/display-drivers/vendor_opensource_display-drivers-peridot-u-oss"
DMSM="$D/msm"
echo "=== build M=msm with DISPLAY_ROOT ==="
make -C "$GITHUB_WORKSPACE/kernel" O="$O" M="$DMSM" \
  DISPLAY_ROOT="$D" \
  modules 2>&1 | tail -30
echo "=== ko ==="
find "$DMSM" -name "*.ko" -exec ls -la {} \; 2>/dev/null
df -h / | tail -1
